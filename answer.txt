# requires: pip install requests beautifulsoup4
import re, requests
from bs4 import BeautifulSoup

usernames = [
    "elephant.225829",  # try 225829 and 245829 if one doesn't exist
    "elephant.245829",
    "zebra.3520519",
    "cat.8629726",
    "panda.66332503"
]

FLAG_RE = re.compile(r"CYBERYUDH\{.*?\}", re.IGNORECASE)

headers = {
    "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko)"
}

def check_user(u):
    url = f"https://www.instagram.com/{u}/"
    try:
        r = requests.get(url, headers=headers, timeout=10)
    except Exception as e:
        print(f"{u}: request error: {e}")
        return None
    if r.status_code != 200:
        print(f"{u}: HTTP {r.status_code}")
        return None
    html = r.text
    # quick flag search in page HTML
    m = FLAG_RE.search(html)
    if m:
        return m.group(0)
    # try to find sharedData script where captions are embedded
    if "window._sharedData" in html:
        mjson = re.search(r"window\._sharedData = (.*?);</script>", html, re.S)
        if mjson:
            jtxt = mjson.group(1)
            m2 = FLAG_RE.search(jtxt)
            if m2:
                return m2.group(0)
    # fallback: parse any visible text
    soup = BeautifulSoup(html, "html.parser")
    text = soup.get_text(" ", strip=True)
    m3 = FLAG_RE.search(text)
    if m3:
        return m3.group(0)
    return None

for u in usernames:
    print("Checking", u)
    flag = check_user(u)
    if flag:
        print("FOUND FLAG:", flag)
        break
else:
    print("Flag not found in those public pages. If accounts are private you will need to authenticate (or try the alternate elephant ID).")
